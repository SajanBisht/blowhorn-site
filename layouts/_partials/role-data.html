{{- /* The data entry for a /for//, /use-cases// or /solutions// page. Call with dict "page" and "kind" (role | use-case | solution). Kind selects the collection through cast-kind.html; the page's params.slug is the entry. */ -}} {{- $kind := .kind -}} {{- $page := .page -}} {{- $meta := partial "cast-kind.html" (dict "kind" $kind) -}} {{- $slug := $page.Params.slug -}} {{- if not $slug }}{{ errorf "no slug param for %s" $page.Path }}{{ end -}} {{- $match := where $meta.collection "slug" $slug -}} {{- if not $match }}{{ errorf "%s: no %s %q for %s" $meta.file $meta.kindName $slug $page.Path }}{{ end -}} {{- return index $match 0 -}}
