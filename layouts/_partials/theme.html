{{- /* Resolve the page theme from data/themes.json. Site default is data.themes.default; a page may set front matter theme: signal|reveille. ?theme= is a token-layer preview in theme-head.html, not a layout change, and is not resolved here. Returns the theme map (id, name, home, colorScheme). */ -}} {{- $data := site.Data.themes -}} {{- $id := $data.default -}} {{- with .Params.theme }}{{ $id = . }}{{ end -}} {{- $match := where $data.themes "id" $id -}} {{- if not $match -}} {{- errorf "unknown theme %q; use one of data/themes.json themes[].id (front matter theme: or the site default)" $id -}} {{- end -}} {{- return index $match 0 -}}
