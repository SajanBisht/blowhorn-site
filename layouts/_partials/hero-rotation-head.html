{{- /* Picks the home page hero before first paint. Put in the page's "head" block. The CSS hides every hero's copy ([data-hero-only]) and art ([data-hero-art-only]) except the one named on ; with no JavaScript those attributes are never set and the default variant shows. The script sets them: ?hero= (1-based) or ?hero= forces a hero, for review; otherwise utc-day uses floor(UTC days since the epoch) modulo the number of heroes, and interval uses floor(now / intervalSeconds). Config comes from data/hero.json via hero-rotation.html. It runs once, at load: the hero never changes under a reader. See docs/hero-variants.md. */ -}} {{- $rot := partial "hero-rotation.html" . -}} {{- $defaultArt := "stage" -}} {{- range $rot.heroes }}{{ if eq .id $rot.default }}{{ $defaultArt = .art }}{{ end }}{{ end -}} {{- $copy := slice (printf `html:not([data-hero]) [data-hero-only="%s"]` $rot.default) -}} {{- range $rot.heroes }}{{ $copy = $copy | append (printf `html[data-hero="%s"] [data-hero-only="%s"]` .id .id) }}{{ end -}} {{- $art := slice (printf `html:not([data-hero]) [data-hero-art-only="%s"]` $defaultArt) -}} {{- range $rot.arts }}{{ $art = $art | append (printf `html[data-hero-art="%s"] [data-hero-art-only="%s"]` . .) }}{{ end -}} {{- $picks := slice -}} {{- range $rot.heroes }}{{ $picks = $picks | append (dict "id" .id "art" .art) }}{{ end }}
