{{- /* One surface pane for an action example. Input: surface (id), data (the surface map), platform, action, actionName. Returns HTML. */ -}} {{- $id := .surface -}} {{- $d := .data -}} {{- $platform := .platform -}} {{- $action := .action -}} {{- $name := .actionName -}} {{- $html := "" -}} {{- if eq $id "app" -}} {{- $path := $d.screenshot -}} {{- $file := printf "static/%s" $path -}} {{- if os.FileExists $file -}} {{- $html = printf `` ($path | relURL) $platform $name -}} {{- else -}} {{- $html = printf `

Screenshot coming soon

Drop a PNG at %s to fill this pane.

` $path -}} {{- end -}} {{- else if eq $id "cli" -}} {{- if eq $d.status "available" -}} {{- $card := partial "code-card.html" (dict "code" $d.command "title" "terminal" "lang" "bash" "prompt" true) -}} {{- $note := "" -}} {{- with $d.notes }}{{ $note = printf `

%s

` . }}{{ end -}} {{- $ref := "" -}} {{- with $d.cliRef }}{{ $ref = printf `

CLI reference

` (. | relURL) }}{{ end -}} {{- $html = printf "%s%s%s" $card $note $ref -}} {{- else -}} {{- $html = printf `

Not in the CLI yet

%s

` ($d.notes | default "This action has no blowhorn command yet.") -}} {{- end -}} {{- else if eq $id "rest" -}} {{- if eq $d.status "available" -}} {{- $html = printf `

%s

CLI reference

` ($d.notes | default "This action is on the Blowhorn REST API.") ((site.GetPage "/docs/reference/cli").RelPermalink) -}} {{- else -}} {{- $html = printf `

Coming soon

%s

` ($d.notes | default "The Blowhorn REST API for this action is not published yet.") -}} {{- end -}} {{- else if eq $id "mcp" -}} {{- $prompt := $d.prompt | default (printf "Do the %s action on %s as me." $name $platform) -}} {{- $card := partial "code-card.html" (dict "code" $prompt "title" "prompt" "lang" "text" "copy" true) -}} {{- if eq $d.status "available" -}} {{- $html = printf `

%s Blowhorn MCP server

%s` ($d.notes | default "Ask an assistant connected to the Blowhorn MCP server.") ((site.GetPage "/ai/mcp").RelPermalink) $card -}} {{- else -}} {{- $html = printf `

Coming soon

%s

%s` ($d.notes | default "Ask an agent connected to the Blowhorn MCP server.") $card -}} {{- end -}} {{- else if eq $id "skill" -}} {{- if eq $d.status "available" -}} {{- $html = printf `

%s

Set up the skill

` ($d.notes | default "The Blowhorn skill covers this action.") ((site.GetPage "/agents/claude-code").RelPermalink) -}} {{- else -}} {{- $html = printf `

Coming soon

%s

` ($d.notes | default "A Blowhorn skill for this action is not available yet.") -}} {{- end -}} {{- else -}} {{- errorf "action-example-pane: unknown surface %q" $id -}} {{- end -}} {{- return $html -}}
